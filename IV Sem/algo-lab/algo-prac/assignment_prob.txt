// Function to swap two integers
void swap(int *a, int *b){
    int temp = *a;
    *a = *b;
    *b = temp;
}

// Function to calculate permutations of job assignments
void permute(int arr[], int n, double * mat[], int l, int r, double *cost, int res[], int * opCount) {
    int i;
    // Base case: if we have a complete permutation
    if (l == r) {
        double tempCost = 0;
        // Calculate the cost of the current permutation
        for (i = 0; i < n ; i++) 
            tempCost += mat[i][arr[i]];
        // Increment operation count for comparisons and assignments
        *opCount = *opCount + n + 1;
        // If the cost is lower than the current minimum, update cost and result
        if (*cost > tempCost) {
            *cost = tempCost;
            for(i=0;i<n;i++) 
                res[i] = arr[i];
            *opCount = *opCount + n;
        }
    }
    else {
        // Recursive case: generate permutations recursively
        for (i = l; i <= r ; i++) {
            // Swap current element with the first element
            swap(&arr[l],&arr[i]);
            // Recur for the rest of the elements
            permute(arr,n,mat,l+1,r,cost,res,opCount);
            // Backtrack: restore the array to its original state
            swap(&arr[l],&arr[i]);
        }
    }
}
